import subprocess
import datetime
import sys
import torch

def sh(command):
    result = subprocess.run(
        command,
        shell=True,
        capture_output=True,
        text=True
    )
    return (
        f"$ {command}\n"
        f"return_code: {result.returncode}\n"
        f"{result.stdout}"
        f"{result.stderr}"
    )

--- key packages ---
{sh("pip list | grep -iE 'transformers|trl|peft|accelerate|bitsandbytes|datasets|soup'")}
"""
